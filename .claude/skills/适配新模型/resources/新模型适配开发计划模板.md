```typescript
  matches: (modelId: string) =>
    modelId === '{model-id}' || modelId === '{别名}',

  buildVideoRequest: async (params: GenerateVideoParams) => {

      }

      endpoint = '/async/{model}-t2v'
      requestData = {
        prompt: params.prompt,
        // 其他参数...
      }

```

**Fal 路由模板：**

```typescript
import { GenerateImageParams } from '@/adapters/base/BaseAdapter'
import { FalModelRoute } from './index'

export const {modelName}Route: FalModelRoute = {
  matches: (modelId: string) =>
    modelId === 'fal-ai/{model}' || modelId === '{model}',

  buildImageRequest: (params: GenerateImageParams) => {
    const images = params.images || []  // 已是 Fal CDN URL

    const submitPath = images.length > 0
      ? 'fal-ai/{model}/edit'  // 图生图端点
      : 'fal-ai/{model}'       // 文生图端点

    const modelId = 'fal-ai/{model}'  // 不含子路径

    const requestData: any = {
      prompt: params.prompt
    }

    // 关键：过滤 'smart' 和 'auto'
    if (params.aspect_ratio &&
        params.aspect_ratio !== 'smart' &&
        params.aspect_ratio !== 'auto') {
      requestData.aspect_ratio = params.aspect_ratio
    }

    if (images.length > 0) {
      requestData.image_url = images[0]
    }

    return { submitPath, modelId, requestData }
  }
}
```

**KIE 路由模板：**

## 附录：供应商特定注意事项

> 按照实际的供应商保留对应的注意实现

### PPIO 适配要点

1. 图片处理：路由收到的 `params.images` 已是 Base64，直接使用
2. 视频处理：路由收到的 `params.video` 已是公开 URL
3. 端点路径：异步任务必须以 `/async/` 开头
4. 状态映射：同时检查 `TASK_STATUS_SUCCEED` 和 `TASK_STATUS_SUCCEEDED`

### Fal 适配要点

1. 图片/视频：路由收到的 `params.images`/`params.videos` 已是 Fal CDN URL
2. 路径区分：`modelId` 不含子路径，`submitPath` 可含子路径
3. 禁止传递：永远不要传递 `'smart'` 或 `'auto'` 给 API
4. 价格转换：使用美元价格乘以 `USD_TO_CNY` 汇率（7.071）
5. OptionsBuilder：`convertToBlob` 必须为 `false`

### KIE 适配要点

1. 图片处理：路由收到的 `params.images` 已是 KIE CDN URL
2. 统一端点：所有模型使用 `/api/v1/jobs/createTask`
3. 智能选项：始终显示，由用户手动选择
4. 本地保存：所有媒体调用 `saveMediaLocally` 并设置 `filePath`
