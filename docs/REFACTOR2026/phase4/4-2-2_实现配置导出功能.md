# 4-2-2 实现配置导出功能

## 目标

实现模型配置和当前参数状态的导出功能，方便分享配置、调试问题、创建预设

## 背景

在开发和调试过程中，经常需要：
- 分享特定的参数配置（如报告 Bug 时附带参数）
- 导出模型配置供其他开发者参考
- 快速创建预设而不用手动填写
- 对比不同模型的配置差异
- 备份当前工作状态

当前没有统一的导出机制，需要手动复制或截图。

export function ExportPanel({ modelId, params }: ExportPanelProps) {
  const [exportType, setExportType] = useState<ExportType>('current-params')
  const [includeDefaults, setIncludeDefaults] = useState(false)
  const [includeSensitive, setIncludeSensitive] = useState(false)

  const handleExport = () => {
    const exportService = new ExportService()
    const data = exportService.export(exportType, modelId, params)

    // 根据用户选择清理数据
    if (!includeDefaults) {
      data.data = cleaner.removeDefaults(data.data, schema)
    }
    if (!includeSensitive) {
      data.data = cleaner.removeSensitive(data.data)
    }

    return data
  }

  return (
    <div className="export-panel">
      <h3>导出配置</h3>

      <div className="export-type-selector">
        <label>导出类型：</label>
        <select value={exportType} onChange={e => setExportType(e.target.value)}>
          <option value="current-params">当前参数</option>
          <option value="model-config">模型配置</option>
          <option value="model-schema">参数 Schema</option>
          <option value="api-request">API 请求</option>
          <option value="preset">预设格式</option>
        </select>
      </div>

      <div className="export-options">
        <label>
          <input
            type="checkbox"
            checked={includeDefaults}
            onChange={e => setIncludeDefaults(e.target.checked)}
          />
          包含默认值
        </label>

        <label>
          <input
            type="checkbox"
            checked={includeSensitive}
            onChange={e => setIncludeSensitive(e.target.checked)}
          />
          包含敏感信息
        </label>
      </div>

      <div className="export-actions">
        <button onClick={() => {
          const data = handleExport()
          exportService.downloadAsJSON(data)
        }}>
          下载 JSON
        </button>

        <button onClick={async () => {
          const data = handleExport()
          await exportService.copyToClipboard(data)
          showToast('已复制到剪贴板')
        }}>
          复制到剪贴板
        </button>
      </div>

      <div className="export-preview">
        <h4>预览</h4>
        <pre>{JSON.stringify(handleExport(), null, 2)}</pre>
      </div>
    </div>
  )
}

    const params = { duration: 5, quality: '720P' }
    const schema = [
      { id: 'duration', default: 5 },
      { id: 'quality', default: '1080P' }

    expect(cleaned).not.toHaveProperty('duration')
    expect(cleaned).toHaveProperty('quality')

## 风险与注意事项

### 风险
- 导出的数据可能包含敏感信息
- Base64 数据可能导致文件过大
- 版本不兼容导致导入失败

### 注意事项
- 默认移除敏感信息（API key、token）
- Base64 数据替换为占位符或移除
- 导出时添加版本号，支持向后兼容
- 限制分享链接长度（URL 长度限制）
- 提供数据验证，拒绝无效导入
- 导出前提示用户检查敏感信息
- 文件名包含时间戳，避免覆盖

### ExportService 完整实现

```typescript
// src/core/export/ExportService.ts

import { registry } from '@/core/ModelRegistry'
import { RequestBuilder } from '@/core/RequestBuilder'

export class ExportService {
  private version = '1.0'

  exportCurrentParams(
    modelId: string,
    params: Record<string, any>
  ): ExportData {
    return this.export('current-params', modelId, params)
  }

  exportModelConfig(modelId: string): ExportData {
    const model = registry.getModel(modelId)
    if (!model) {
      throw new Error(`Model not found: ${modelId}`)
    }

    return this.export('model-config', modelId, model)
  }

  exportModelSchema(modelId: string): ExportData {
    const schema = registry.getSchema(modelId)
    return this.export('model-schema', modelId, schema)
  }

  exportAPIRequest(
    modelId: string,
    params: Record<string, any>,
    context: any
  ): ExportData {
    const builder = new RequestBuilder()
    const request = builder.buildRequest(modelId, params, context)
    return this.export('api-request', modelId, request)
  }

  exportAsPreset(
    modelId: string,
    params: Record<string, any>,
    presetName: string
  ): ExportData {
    const presetData = {
      name: presetName,
      modelId,
      params
    }
    return this.export('preset', modelId, presetData)
  }

  export(type: ExportType, modelId: string, data: any): ExportData {
    return {
      version: this.version,
      type,
      timestamp: Date.now(),
      modelId,
      data,
      metadata: {
        appVersion: import.meta.env.VITE_APP_VERSION || 'dev',
        platform: navigator.platform,
        userAgent: navigator.userAgent
      }
    }
  }

  downloadAsJSON(data: ExportData, filename?: string): void {
    const json = JSON.stringify(data, null, 2)
    const blob = new Blob([json], { type: 'application/json' })
    const url = URL.createObjectURL(blob)

    const a = document.createElement('a')
    a.href = url
    a.download = filename || this.generateFilename(data)
    document.body.appendChild(a)
    a.click()
    document.body.removeChild(a)
    URL.revokeObjectURL(url)
  }

  async copyToClipboard(data: ExportData): Promise<void> {
    const json = JSON.stringify(data, null, 2)
    await navigator.clipboard.writeText(json)
  }

  private generateFilename(data: ExportData): string {
    const date = new Date(data.timestamp).toISOString().slice(0, 10)
    return `henji-${data.type}-${data.modelId}-${date}.json`
  }
}
```

### ParamCleaner 实现

```typescript
// src/core/export/ParamCleaner.ts

export class ParamCleaner {
  clean(
    params: Record<string, any>,
    options: CleanOptions = {}
  ): Record<string, any> {
    let result = { ...params }

    if (options.removeDefaults && options.schema) {
      result = this.removeDefaults(result, options.schema)
    }

    if (options.removeEmpty) {
      result = this.removeEmpty(result)
    }

    if (options.removeSensitive) {
      result = this.removeSensitive(result)
    }

    if (options.removeBase64) {
      result = this.removeBase64(result)
    }

    return result
  }

  removeDefaults(
    params: Record<string, any>,
    schema: ParamDef[]
  ): Record<string, any> {
    const result: Record<string, any> = {}

    Object.entries(params).forEach(([key, value]) => {
      const paramDef = schema.find(p => p.id === key)
      if (!paramDef || paramDef.default !== value) {
        result[key] = value
      }
    })

    return result
  }

  removeEmpty(params: Record<string, any>): Record<string, any> {
    const result: Record<string, any> = {}

    Object.entries(params).forEach(([key, value]) => {
      if (value !== null && value !== undefined && value !== '') {
        if (Array.isArray(value) && value.length === 0) return
        if (typeof value === 'object' && Object.keys(value).length === 0) return
        result[key] = value
      }
    })

    return result
  }

  removeSensitive(params: Record<string, any>): Record<string, any> {
    const sensitiveKeys = ['apiKey', 'token', 'password', 'secret']
    const result = { ...params }

    Object.keys(result).forEach(key => {
      if (sensitiveKeys.some(sk => key.toLowerCase().includes(sk))) {
        result[key] = '[REDACTED]'
      }
    })

    return result
  }

  removeBase64(params: Record<string, any>): Record<string, any> {
    const result: Record<string, any> = {}

    Object.entries(params).forEach(([key, value]) => {
      if (typeof value === 'string' && value.startsWith('data:')) {
        result[key] = '[Base64 Data]'
      } else if (Array.isArray(value)) {
        result[key] = value.map(v =>
          typeof v === 'string' && v.startsWith('data:')
            ? '[Base64 Data]'
            : v
        )
      } else {
        result[key] = value
      }
    })

    return result
  }
}
```
