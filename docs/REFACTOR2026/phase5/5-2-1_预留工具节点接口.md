# 5-2-1 预留工具节点接口

## 目标

设计并实现常用工具节点的接口定义，为画布模式提供图片处理、文本处理等辅助功能，但不实现完整的工具节点 UI。

## 背景

除了模型节点，画布模式还需要一些辅助工具节点来处理数据流：
- **图片处理**：裁剪、缩放、格式转换
- **文本处理**：模板、拼接、提取
- **逻辑控制**：条件分支、循环
- **数据转换**：类型转换、映射

当前阶段（Phase 5）的目标是**设计接口和基础实现**，为未来的画布 UI 做准备。

```typescript
class ToolNodeRegistry {
  private static instance: ToolNodeRegistry
  private nodes: Map<string, ToolNodeDefinition> = new Map()

  static getInstance(): ToolNodeRegistry {
    if (!ToolNodeRegistry.instance) {
      ToolNodeRegistry.instance = new ToolNodeRegistry()
    }
    return ToolNodeRegistry.instance
  }

  register(def: ToolNodeDefinition): void {
    if (this.nodes.has(def.type)) {
      throw new Error(`Tool node type already exists: ${def.type}`)
    }
    this.nodes.set(def.type, def)
  }

  get(type: string): ToolNodeDefinition | undefined {
    return this.nodes.get(type)
  }

  listByCategory(category: string): ToolNodeDefinition[] {
    return Array.from(this.nodes.values())
      .filter(node => node.category === category)
  }
}
```

interface ToolNodeDefinition {
  // 基础信息
  type: string                    // 工具类型（唯一标识）
  name: I18nText                  // 显示名称
  description?: I18nText          // 描述
  icon?: string                   // 图标
  category: ToolNodeCategory      // 分类

  // 端口定义
  inputs: InputPort[]
  outputs: OutputPort[]

  // 执行函数
  execute: (inputs: Record<string, any>) => Promise<NodeOutput> | NodeOutput

  // 配置 Schema（可选）
  configSchema?: ParamDef[]
}

type ToolNodeCategory =
  | 'image-processing'      // 图片处理
  | 'text-processing'       // 文本处理
  | 'data-conversion'       // 数据转换
  | 'logic-control'         // 逻辑控制
  | 'utility'               // 实用工具
```

export const promptTemplateNode = defineToolNode({
  type: 'prompt-template',
  name: { zh: '提示词模板', en: 'Prompt Template' },
  description: {
    zh: '使用模板生成提示词，支持变量替换',
    en: 'Generate prompt using template with variable substitution'
  },
  category: 'text-processing',

  inputs: [
    {
      id: 'template',
      name: { zh: '模板', en: 'Template' },
      type: 'string',
      required: true,
      default: '一个{{subject}}，{{style}}风格，{{details}}'
    },
    {
      id: 'variables',
      name: { zh: '变量', en: 'Variables' },
      type: 'object',
      required: false,
      default: {}
    }
  ],

  outputs: [
    {
      id: 'output',
      name: { zh: '生成的提示词', en: 'Generated Prompt' },
      type: 'string'
    }
  ],

  execute: (inputs) => {
    const { template, variables = {} } = inputs
    let result = template

    // 简单的变量替换
    for (const [key, value] of Object.entries(variables)) {
      result = result.replace(new RegExp(`{{${key}}}`, 'g'), String(value))
    }

    return { output: result }
  }
})
```

export const convertTypeNode = defineToolNode({
  type: 'convert-type',

  category: 'data-conversion',

  inputs: [
    {
      id: 'input',
      name: { zh: '输入值', en: 'Input Value' },
      type: 'any',
      required: true
    },
    {
      id: 'targetType',
      name: { zh: '目标类型', en: 'Target Type' },

    }
  ],

  outputs: [
    {
      id: 'output',
      name: { zh: '转换后值', en: 'Converted Value' },
      type: 'any'
    }
  ],

  execute: (inputs) => {
    const { input, targetType } = inputs

    switch (targetType) {
      case 'string':
        return { output: String(input) }
      case 'number':
        return { output: Number(input) }
      case 'boolean':
        return { output: Boolean(input) }
      default:
        return { output: input }
    }
  }
})

```typescript
export function defineToolNode(def: ToolNodeDefinition): ToolNodeDefinition {
  // 验证定义
  if (!def.type) throw new Error('Tool node type is required')
  if (!def.name) throw new Error('Tool node name is required')
  if (!def.execute) throw new Error('Tool node execute function is required')

  // 自动注册
  const registry = ToolNodeRegistry.getInstance()
  registry.register(def)

  return def
}
```

## 风险与注意事项

### 风险
- 图片处理节点的实现可能依赖浏览器 API（Canvas），在 Tauri 环境中需要验证
- 工具节点的数量和功能范围难以预估

### 注意事项
- 工具节点的执行应该是纯函数，不依赖外部状态
- 图片处理节点暂时只提供接口，完整实现在画布模式开发时完成
- 考虑异步执行的工具节点（如网络请求）
- 工具节点应该有良好的错误处理
- 预留自定义工具节点的扩展机制
