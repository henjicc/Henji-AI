```typescript
// 端口数据类型
export type PortDataType =
  | 'string'      // 文本（如 prompt）
  | 'number'      // 数字
  | 'boolean'     // 布尔值
  | 'image'       // 图片（URL 或 base64）
  | 'video'       // 视频（URL）
  | 'audio'       // 音频（URL）
  | 'array'       // 数组
  | 'object'      // 对象
  | 'any'         // 任意类型

// 输入端口接口
export interface InputPort {
  id: string
  name: I18nText
  type: PortDataType
  required: boolean
  default?: any
  description?: I18nText
}

// 输出端口接口
export interface OutputPort {
  id: string
  name: I18nText
  type: PortDataType
  description?: I18nText
}
```

**设计亮点**:
- 支持 9 种端口数据类型，覆盖所有常见场景
- 输入端口支持必需标记和默认值
- 完整的国际化支持

### 2. 创建模型节点接口

**文件路径**: `src/core/types/ModelNode.ts`

**核心接口**:

```typescript
export interface ModelNode {
  id: string                      // 节点实例 ID
  type: 'model'                   // 节点类型
  modelId: string                 // 对应的模型 ID

  meta: {
    name: I18nText
    description?: I18nText
    icon?: string
    category: 'image' | 'video' | 'audio'
    tags?: string[]
  }

  inputs: InputPort[]             // 输入端口
  outputs: OutputPort[]           // 输出端口
  position?: { x: number; y: number }
  execute: NodeExecutor           // 执行函数
}

export interface NodeOutput {
  [portId: string]: any
  _metadata?: {
    taskId?: string
    status?: string
    error?: string
  }
}

export interface ExecutionContext {
  onProgress?: (status: ProgressStatus) => void
  signal?: AbortSignal
}
```

**设计亮点**:
- 节点元数据与 ModelDefinition 保持一致
- 支持画布位置信息
- 执行器支持进度回调和取消信号
- 输出包含元数据，便于错误处理
