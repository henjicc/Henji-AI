## 二、类型定义详解

### 1. I18nText（国际化文本）

**文件：** `src/core/types/I18nText.ts`

```typescript
export type I18nText = string | {
  zh: string
  en: string
}
```

**功能：**
- 支持纯字符串或多语言对象
- 提供 `getI18nText()` 工具函数

**使用示例：**
```typescript
const name1: I18nText = "Nano Banana"
const name2: I18nText = { zh: "即梦图片", en: "Seedream" }

getI18nText(name2, 'zh')  // "即梦图片"
getI18nText(name2, 'en')  // "Seedream"
```

### 2. EndpointConfig（端点配置）

**文件：** `src/core/types/EndpointConfig.ts`

```typescript
export type EndpointConfig =
  | string  // 固定端点
  | {
      rules?: EndpointRule[]
      default?: string
      selector?: (params: Record<string, any>) => string
    }
```

**支持三种配置方式：**

#### 方式1：固定端点

```typescript
endpoints: '/api/v1/generate'
```

#### 方式2：规则选择

```typescript
endpoints: {
  rules: [
    { when: { mode: 'text-to-video' }, endpoint: '/api/v1/text-to-video' },
    { when: { mode: 'image-to-video' }, endpoint: '/api/v1/image-to-video' }
  ],
  default: '/api/v1/generate'
}
```

#### 方式3：函数选择

```typescript
endpoints: {
  selector: (params) => {
    if (params.mode === 'text-to-video') return '/api/v1/text-to-video'
    return '/api/v1/generate'
  }
}
```

### 3. RequestConfig（请求构建）

**文件：** `src/core/types/RequestConfig.ts`

```typescript
export interface RequestConfig {
  base?: Record<string, string>
  preprocess?: (params: Record<string, any>) => Record<string, any>
  builder?: (params: Record<string, any>) => Record<string, any>
}
```

**三种请求构建方式：**

#### 方式1：基础字段映射

```typescript
request: {
  base: {
    prompt: 'input.prompt',
    size: 'options.resolution',
    num_images: 'options.count'
  }
}
```

#### 方式2：预处理函数

```typescript
request: {
  base: { ... },
  preprocess: (params) => {
    return {
      ...params,
      prompt: params.prompt.trim()
    }
  }
}
```

#### 方式3：自定义构建器

```typescript
request: {
  builder: (params) => {
    return {
      model: 'gpt-4',
      messages: [{ role: 'user', content: params.prompt }]
    }
  }
}
```

### 4. PricingConfig（价格配置）

**文件：** `src/core/types/PricingConfig.ts`

```typescript
export interface PricingConfig {
  currency: Currency
  fixed?: number
  calculator?: (params: Record<string, any>) => number
  description?: string
}
```

**支持两种价格模式：**

#### 固定价格

```typescript
pricing: {
  currency: '¥',
  fixed: 0.1
}
```

#### 动态计算

```typescript
pricing: {
  currency: '$',
  calculator: (params) => {
    const duration = params.duration || 5
    const pricePerSecond = 0.02
    return duration * pricePerSecond
  },
  description: '按视频时长计费：$0.02/秒'
}
```

### 5. ModelDefinition（核心接口）

**文件：** `src/core/types/ModelDefinition.ts`

```typescript
export interface ModelDefinition {
  meta: ModelMeta
  params: ParamDef[]
  linkages?: Linkage[]
  endpoints: EndpointConfig
  request?: RequestConfig
  pricing: PricingConfig
}
```

#### ModelMeta（元数据）

```typescript
export interface ModelMeta {
  id: string
  provider: ProviderId
  type: ModelType
  name: I18nText
  description?: I18nText
  tags?: ModelTag[]
  icon?: string
  polling?: {
    interval: number
    maxAttempts: number
    expectedAttempts?: number
  }
  aliases?: string[]
}
```

**字段说明：**

| 字段 | 类型 | 必需 | 说明 |
|------|------|------|------|
| `id` | string | ✅ | 模型唯一标识 |
| `provider` | ProviderId | ✅ | Provider ID |
| `type` | ModelType | ✅ | 模型类型（image/video/audio） |
| `name` | I18nText | ✅ | 模型名称 |
| `description` | I18nText | ❌ | 模型描述 |
| `tags` | ModelTag[] | ❌ | 模型标签（来自 Phase 0-2-1） |
| `icon` | string | ❌ | 图标 URL |
| `polling` | object | ❌ | 轮询配置 |
| `aliases` | string[] | ❌ | 模型别名 |

```typescript
export const nanoBananaExample: ModelDefinition = {
  meta: {
    id: 'nano-banana',

    type: 'image',
    name: { zh: 'Nano Banana', en: 'Nano Banana' },

      'text-to-image',
      'image-to-image',
      'supports-image-editing',
      'supports-multi-image',
      'no-resolution-panel',
      'provider-fal'
    ],
    polling: {
      interval: 3000,
      maxAttempts: 100,
      expectedAttempts: 30
    },
    aliases: ['fal-ai-nano-banana']
  },

  params: [],  // TODO: Phase 1-1-2

  linkages: [],  // TODO: Phase 1-1-3

  endpoints: {
    rules: [
      { when: { hasImage: true }, endpoint: '/fal-ai/nano-banana/image-to-image' },
      { when: { hasImage: false }, endpoint: '/fal-ai/nano-banana/text-to-image' }
    ],
    default: '/fal-ai/nano-banana'
  },

  request: {
    base: {
      prompt: 'input.prompt',
      image_url: 'input.imageUrl',
      num_images: 'options.numImages',
      aspect_ratio: 'options.aspectRatio'
    },
    preprocess: (params) => {
      const processed = { ...params }
      if (Array.isArray(params.images) && params.images.length > 0) {
        processed.image_url = params.images[0]
      }
      return processed
    }
  },

  pricing: {
    currency: '¥',
    fixed: 0.1
  }
}
```

### 示例3：Kling 2.6 Pro（视频模型）

```typescript
export const kling26ProExample: ModelDefinition = {
  meta: {
    id: 'kling-2.6-pro',
    provider: 'ppio',
    type: 'video',
    name: { zh: '可灵 2.6 Pro', en: 'Kling 2.6 Pro' },
    tags: [
      'text-to-video',
      'image-to-video',
      'supports-motion-control',
      'supports-audio-generation',
      'multi-mode-switch'
    ]
  },

  params: [],

  endpoints: {
    rules: [
      { when: { mode: 'text-to-video' }, endpoint: '/api/ppio/kling-2.6-pro/text-to-video' },
      { when: { mode: 'image-to-video' }, endpoint: '/api/ppio/kling-2.6-pro/image-to-video' },
      { when: { mode: 'motion-control' }, endpoint: '/api/ppio/kling-2.6-pro/motion-control' }
    ],
    default: '/api/ppio/kling-2.6-pro'
  },

  request: {
    builder: (params) => {
      const request: Record<string, any> = {
        prompt: params.prompt,
        mode: params.mode || 'text-to-video'
      }

      if (params.mode === 'image-to-video' || params.mode === 'motion-control') {
        request.images = params.images
      }

      if (params.mode === 'motion-control') {
        request.reference_video = params.referenceVideo
      }

      return request
    }
  },

  pricing: {
    currency: '¥',
    calculator: (params) => {
      const duration = params.duration || 5
      let pricePerSecond = params.mode === 'motion-control' ? 0.5 : 0.3
      if (params.generateAudio) pricePerSecond += 0.1
      return duration * pricePerSecond
    }
  }
}
```
