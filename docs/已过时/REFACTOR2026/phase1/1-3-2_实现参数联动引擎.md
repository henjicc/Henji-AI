# 1-3-2 实现参数联动引擎

## 目标

实现参数联动系统，支持参数间的依赖、过滤、自动切换等复杂交互

## 背景

当前问题：
- 联动逻辑分散在组件中（如 MediaGenerator/index.tsx 中的 useEffect）
- 每个联动关系需要手动编写代码
- 难以维护和扩展

新架构目标：
- 配置驱动的联动系统
- 支持多种联动效果（reset、filterOptions、autoSwitch 等）
- 自动执行顺序管理
- 防止循环联动

联动类型（REFACTOR_PLAN.md 3.3 节）：
- reset - 重置目标参数
- filterOptions - 过滤选项
- filterRange - 过滤范围
- setValue - 设置值
- autoSwitch - 条件自动切换
- disable - 禁用参数
- hide - 隐藏参数
- custom - 自定义处理

## 风险与注意事项

### 风险
- 复杂的联动链可能导致性能问题
- 循环联动检测可能漏掉边缘情况
- 字符串表达式求值存在安全风险

### 注意事项
- 使用不可变更新避免副作用
- 限制联动执行深度（如 10 层）
- 条件求值应使用安全沙箱（避免 eval）
- 记录详细的执行日志便于调试
- 考虑使用 Web Worker 执行复杂联动
- autoSwitch 应记录原始值，用于恢复
- 批量更新时只触发一次联动
- 提供联动禁用开关（测试用）

```typescript
// src/core/linkage/LinkageEngine.ts

import type { Linkage, ParamDef } from '../types'
import { executeEffect } from './effects'
import { evaluateCondition } from './evaluator'

export class LinkageEngine {
  private linkages: Linkage[] = []
  private executionDepth = 0
  private maxDepth = 10

  constructor(linkages: Linkage[]) {
    this.linkages = this.sortLinkages(linkages)
  }

  /**
   * 按效果类型和优先级排序
   */
  private sortLinkages(linkages: Linkage[]): Linkage[] {
    const effectOrder: Record<string, number> = {
      reset: 1,
      setValue: 2,
      autoSwitch: 3,
      filterOptions: 4,
      filterRange: 5,
      disable: 6,
      hide: 7,
      custom: 8
    }

    return [...linkages].sort((a, b) => {
      const priorityA = a.priority ?? effectOrder[a.effect] ?? 999
      const priorityB = b.priority ?? effectOrder[b.effect] ?? 999
      return priorityA - priorityB
    })
  }

  /**
   * 执行联动
   */
  execute(
    trigger: string,
    params: Record<string, any>,
    schema: ParamDef[]
  ): Record<string, any> {
    // 防止循环联动
    if (this.executionDepth >= this.maxDepth) {
      console.warn('[Linkage] Max depth reached, stopping execution')
      return params
    }

    this.executionDepth++

    try {
      let result = { ...params }
      const triggerValue = params[trigger]

      for (const linkage of this.linkages) {
        // 检查是否应该触发
        if (!this.shouldTrigger(linkage, trigger)) {
          continue
        }

        // 执行效果
        const changes = executeEffect(linkage, triggerValue, result, schema)

        if (changes) {
          result = { ...result, ...changes }
        }
      }

      return result
    } finally {
      this.executionDepth--
    }
  }

  /**
   * 判断是否应该触发联动
   */
  private shouldTrigger(linkage: Linkage, trigger: string): boolean {
    if (Array.isArray(linkage.trigger)) {
      return linkage.trigger.includes(trigger)
    }
    return linkage.trigger === trigger
  }
}
```

```typescript
// src/core/linkage/effects/index.ts

import type { Linkage, ParamDef } from '../../types'

export function executeEffect(
  linkage: Linkage,
  triggerValue: any,
  params: Record<string, any>,
  schema: ParamDef[]
): Record<string, any> | null {
  switch (linkage.effect) {
    case 'reset':
      return executeResetEffect(linkage, schema)

    case 'setValue':
      return executeSetValueEffect(linkage, triggerValue, params)

    case 'autoSwitch':
      return executeAutoSwitchEffect(linkage, triggerValue, params)

    case 'filterOptions':
      // filterOptions 不修改参数，只修改可选项
      // 在 getFilteredOptions 中处理
      return null

    case 'custom':
      return linkage.handler?.(triggerValue, params) || null

    default:
      return null
  }
}

function executeResetEffect(
  linkage: Linkage,
  schema: ParamDef[]
): Record<string, any> {
  const changes: Record<string, any> = {}
  const targets = linkage.targets || (linkage.target ? [linkage.target] : [])

  for (const targetId of targets) {
    const paramDef = schema.find(p => p.id === targetId)
    if (paramDef) {
      changes[targetId] = paramDef.default
    }
  }

  return changes
}

function executeSetValueEffect(
  linkage: Linkage,
  triggerValue: any,
  params: Record<string, any>
): Record<string, any> | null {
  if (!linkage.target) return null

  const value = typeof linkage.value === 'function'
    ? linkage.value(triggerValue, params)
    : linkage.value

  return { [linkage.target]: value }
}

function executeAutoSwitchEffect(
  linkage: Linkage,
  triggerValue: any,
  params: Record<string, any>
): Record<string, any> | null {
  if (!linkage.target || !linkage.condition) return null

  const shouldSwitch = linkage.condition(triggerValue, params)

  if (shouldSwitch) {
    const value = typeof linkage.value === 'function'
      ? linkage.value(triggerValue, params)
      : linkage.value

    return { [linkage.target]: value }
  } else if (!linkage.noRestore) {
    // 恢复默认值
    // 需要从 Schema 获取
    return null // 暂时返回 null，在 Hook 中处理
  }

  return null
}
```

/**
 * 安全求值条件表达式
 */
export function evaluateCondition(
  condition: string | ((params: any) => boolean),
  params: Record<string, any>
): boolean {
  if (typeof condition === 'function') {
    return condition(params)
  }

  // 字符串表达式求值（简化版，生产环境应使用 jsep + 安全求值器）
  try {
    // 创建安全上下文
    const context = new Map(Object.entries(params))

    // 简单的表达式求值（仅支持基本比较）
    // 生产环境应使用 jsep 或其他安全解析器
    const fn = new Function('params', `with(params) { return ${condition} }`)
    return fn(params)
  } catch (error) {
    console.error('[Evaluator] Failed to evaluate condition:', condition, error)
    return false
  }
}
