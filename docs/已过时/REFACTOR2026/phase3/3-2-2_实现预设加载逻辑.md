# 3-2-2 实现预设加载逻辑

## 目标

实现智能预设加载机制，支持参数验证、版本兼容、部分应用和冲突处理。

## 背景

### 当前问题

预设加载缺乏智能处理：
1. 没有参数验证（可能应用无效参数）
2. 没有版本兼容机制（旧预设在新版本可能失败）
3. 全量覆盖（无法部分应用预设）
4. 没有冲突检测（用户修改的参数被覆盖）

### 新架构方案

**核心功能**：
1. **参数验证**：只应用模型支持的参数
2. **智能过滤**：全局预设只应用通用参数
3. **部分应用**：支持选择性应用参数
4. **冲突检测**：提示用户将被覆盖的参数
5. **回退支持**：可撤销预设应用

```typescript
/**
 * 预设应用模式
 */
export type PresetApplyMode =
  | 'full'      // 完全应用（覆盖所有参数）
  | 'partial'   // 部分应用（只应用选中的参数）
  | 'merge'     // 合并应用（只应用未修改的参数）

/**
 * 预设应用选项
 */
export interface PresetApplyOptions {
  /** 应用模式 */
  mode: PresetApplyMode

  /** 部分应用时，指定要应用的参数 ID */
  paramIds?: string[]

  /** 是否跳过冲突检测 */
  skipConflictCheck?: boolean

  /** 是否保存到历史（用于撤销） */
  saveToHistory?: boolean
}

/**
 * 预设应用结果
 */
export interface PresetApplyResult {
  /** 成功应用的参数 */
  applied: AppliedParam[]

  /** 被忽略的参数 */
  ignored: IgnoredParam[]

  /** 冲突的参数 */
  conflicts: ConflictParam[]

  /** 验证失败的参数 */
  invalid: InvalidParam[]
}

/**
 * 应用的参数信息
 */
export interface AppliedParam {
  id: string
  oldValue: any
  newValue: any
}

/**
 * 忽略的参数信息
 */
export interface IgnoredParam {
  id: string
  value: any
  reason: 'not-in-schema' | 'global-preset-filter' | 'not-selected'
}

/**
 * 冲突的参数信息
 */
export interface ConflictParam {
  id: string
  currentValue: any
  presetValue: any
  defaultValue: any
  isModified: boolean  // 用户是否修改过
}

/**
 * 无效的参数信息
 */
export interface InvalidParam {
  id: string
  value: any
  reason: string
}

/**
 * 预设应用历史（用于撤销）
 */
export interface PresetApplyHistory {
  timestamp: number
  presetId: string
  presetName: string
  previousParams: Record<string, any>
  appliedParams: Record<string, any>
}
```

```typescript
import { ParamDef } from '@/core/types/ParamDef'

/**
 * 参数验证器
 */
export class ParamValidator {
  /**
   * 验证参数值是否有效
   */
  static validate(paramDef: ParamDef, value: any): ValidationResult {
    const errors: string[] = []

    // 类型检查
    if (!this.validateType(paramDef.valueType, value)) {
      errors.push(`Type mismatch: expected ${paramDef.valueType}, got ${typeof value}`)
    }

    // 范围检查（数字类型）
    if (paramDef.component === 'number' || paramDef.component === 'slider') {
      if (paramDef.min !== undefined && value < paramDef.min) {
        errors.push(`Value ${value} is less than minimum ${paramDef.min}`)
      }
      if (paramDef.max !== undefined && value > paramDef.max) {
        errors.push(`Value ${value} is greater than maximum ${paramDef.max}`)
      }
    }

    // 选项检查（下拉框、单选等）
    if (paramDef.options && paramDef.component === 'dropdown') {
      const validValues = paramDef.options.map(opt => opt.value)
      if (!validValues.includes(value)) {
        errors.push(`Value ${value} is not in valid options: ${validValues.join(', ')}`)
      }
    }

    // 自定义验证
    if (paramDef.validate) {
      const customError = paramDef.validate(value)
      if (customError) {
        errors.push(customError)
      }
    }

    return {
      valid: errors.length === 0,
      errors
    }
  }

  /**
   * 类型验证
   */
  private static validateType(expectedType: string, value: any): boolean {
    switch (expectedType) {
      case 'string':
        return typeof value === 'string'
      case 'number':
        return typeof value === 'number' && !isNaN(value)
      case 'boolean':
        return typeof value === 'boolean'
      case 'array':
        return Array.isArray(value)
      case 'object':
        return typeof value === 'object' && value !== null && !Array.isArray(value)
      default:
        return true
    }
  }
}

interface ValidationResult {
  valid: boolean
  errors: string[]
}
```

### 步骤 3: 实现预设加载器

**文件**: `src/services/presets/PresetLoader.ts`

```typescript
import { Preset } from '@/core/types/Preset'
import {
  PresetApplyOptions,
  PresetApplyResult,
  AppliedParam,
  IgnoredParam,
  ConflictParam,
  InvalidParam,
  PresetApplyHistory
} from '@/core/types/PresetLoader'
import { registry } from '@/core/ModelRegistry'
import { ParamValidator } from '@/core/validation/ParamValidator'

/**
 * 预设加载器
 */
export class PresetLoader {
  private modelId: string
  private currentParams: Record<string, any>
  private defaultParams: Record<string, any>
  private applyHistory: PresetApplyHistory[] = []

  constructor(modelId: string, currentParams: Record<string, any>) {
    this.modelId = modelId
    this.currentParams = currentParams
    this.defaultParams = this.getDefaultParams()
  }

  /**
   * 应用预设
   */
  apply(preset: Preset, options: PresetApplyOptions): PresetApplyResult {
    const result: PresetApplyResult = {
      applied: [],
      ignored: [],
      conflicts: [],
      invalid: []
    }

    const schema = registry.getSchema(this.modelId)
    const schemaMap = new Map(schema.map(p => [p.id, p]))

    // 1. 过滤和验证参数
    for (const [paramId, value] of Object.entries(preset.params)) {
      const paramDef = schemaMap.get(paramId)

      // 检查参数是否存在于 schema
      if (!paramDef) {
        result.ignored.push({
          id: paramId,
          value,
          reason: 'not-in-schema'
        })
        continue
      }

      // 全局预设过滤：只应用通用参数
      if (preset.modelId === null && !this.isUniversalParam(paramId)) {
        result.ignored.push({
          id: paramId,
          value,
          reason: 'global-preset-filter'
        })
        continue
      }

      // 部分应用模式：只应用选中的参数
      if (options.mode === 'partial' && options.paramIds) {
        if (!options.paramIds.includes(paramId)) {
          result.ignored.push({
            id: paramId,
            value,
            reason: 'not-selected'
          })
          continue
        }
      }

      // 验证参数值
      const validation = ParamValidator.validate(paramDef, value)
      if (!validation.valid) {
        result.invalid.push({
          id: paramId,
          value,
          reason: validation.errors.join('; ')
        })
        continue
      }

      // 冲突检测
      const currentValue = this.currentParams[paramId]
      const defaultValue = this.defaultParams[paramId]
      const isModified = currentValue !== defaultValue

      if (!options.skipConflictCheck && isModified && currentValue !== value) {
        result.conflicts.push({
          id: paramId,
          currentValue,
          presetValue: value,
          defaultValue,
          isModified
        })

        // 合并模式：跳过已修改的参数
        if (options.mode === 'merge') {
          continue
        }
      }

      // 记录应用的参数
      result.applied.push({
        id: paramId,
        oldValue: currentValue,
        newValue: value
      })
    }

    // 2. 保存到历史（用于撤销）
    if (options.saveToHistory) {
      this.saveToHistory(preset, result.applied)
    }

    return result
  }

  /**
   * 撤销上次预设应用
   */
  undo(): Record<string, any> | null {
    const lastApply = this.applyHistory.pop()
    if (!lastApply) {
      return null
    }

    return lastApply.previousParams
  }

  /**
   * 获取应用历史
   */
  getHistory(): PresetApplyHistory[] {
    return [...this.applyHistory]
  }

  /**
   * 清除历史
   */
  clearHistory(): void {
    this.applyHistory = []
  }

  /**
   * 检查是否为通用参数
   */
  private isUniversalParam(paramId: string): boolean {
    // 通用参数列表（适用于所有模型）
    const universalParams = [
      'prompt',
      'negativePrompt',
      'seed'
    ]
    return universalParams.includes(paramId)
  }

  /**
   * 获取默认参数
   */
  private getDefaultParams(): Record<string, any> {
    const schema = registry.getSchema(this.modelId)
    const defaults: Record<string, any> = {}

    for (const param of schema) {
      defaults[param.id] = param.default
    }

    return defaults
  }

  /**
   * 保存到历史
   */
  private saveToHistory(preset: Preset, applied: AppliedParam[]): void {
    const previousParams: Record<string, any> = {}
    const appliedParams: Record<string, any> = {}

    for (const param of applied) {
      previousParams[param.id] = param.oldValue
      appliedParams[param.id] = param.newValue
    }

    this.applyHistory.push({
      timestamp: Date.now(),
      presetId: preset.id,
      presetName: preset.name,
      previousParams,
      appliedParams
    })

    // 限制历史记录数量
    if (this.applyHistory.length > 10) {
      this.applyHistory.shift()
    }
  }
}
```

### 步骤 4: 更新 usePresetLoader Hook

**文件**: `src/hooks/usePresetLoader.ts`

```typescript
import { useCallback, useState, useRef } from 'react'
import { Preset } from '@/core/types/Preset'
import { PresetApplyOptions, PresetApplyResult } from '@/core/types/PresetLoader'
import { useModelParams } from './useModelParams'
import { PresetLoader } from '@/services/presets/PresetLoader'

/**
 * 预设加载 Hook（增强版）
 */
export function usePresetLoader(currentModelId: string) {
  const { setParams, params, setParam } = useModelParams(currentModelId)
  const [lastApplyResult, setLastApplyResult] = useState<PresetApplyResult | null>(null)
  const loaderRef = useRef<PresetLoader | null>(null)

  // 初始化或更新加载器
  if (!loaderRef.current || loaderRef.current.modelId !== currentModelId) {
    loaderRef.current = new PresetLoader(currentModelId, params)
  }

  /**
   * 应用预设
   */
  const applyPreset = useCallback((preset: Preset, options?: Partial<PresetApplyOptions>) => {
    const fullOptions: PresetApplyOptions = {
      mode: 'full',
      skipConflictCheck: false,
      saveToHistory: true,
      ...options
    }

    const loader = new PresetLoader(currentModelId, params)
    const result = loader.apply(preset, fullOptions)

    // 应用参数
    const newParams: Record<string, any> = {}
    for (const applied of result.applied) {
      newParams[applied.id] = applied.newValue
    }

    if (Object.keys(newParams).length > 0) {
      setParams(prev => ({ ...prev, ...newParams }))
    }

    // 保存结果
    setLastApplyResult(result)
    loaderRef.current = loader

    return result
  }, [currentModelId, params, setParams])

  /**
   * 撤销上次应用
   */
  const undoPreset = useCallback(() => {
    if (!loaderRef.current) {
      return false
    }

    const previousParams = loaderRef.current.undo()
    if (previousParams) {
      setParams(prev => ({ ...prev, ...previousParams }))
      return true
    }

    return false
  }, [setParams])

  /**
   * 从当前参数创建预设
   */
  const createPresetFromCurrent = useCallback((
    name: string,
    description?: string,
    selectedParamIds?: string[]
  ) => {
    let presetParams = { ...params }

    // 如果指定了参数，只包含这些参数
    if (selectedParamIds && selectedParamIds.length > 0) {
      presetParams = {}
      for (const paramId of selectedParamIds) {
        if (paramId in params) {
          presetParams[paramId] = params[paramId]
        }
      }
    }

    return {
      name,
      description,
      modelId: currentModelId,
      params: presetParams
    }
  }, [currentModelId, params])

  /**
   * 预览预设应用结果（不实际应用）
   */
  const previewPreset = useCallback((preset: Preset, options?: Partial<PresetApplyOptions>) => {
    const fullOptions: PresetApplyOptions = {
      mode: 'full',
      skipConflictCheck: false,
      saveToHistory: false,
      ...options
    }

    const loader = new PresetLoader(currentModelId, params)
    return loader.apply(preset, fullOptions)
  }, [currentModelId, params])

  return {
    applyPreset,
    undoPreset,
    createPresetFromCurrent,
    previewPreset,
    lastApplyResult,
    canUndo: loaderRef.current?.getHistory().length > 0
  }
}
```

### 步骤 5: 实现预设应用 UI

**文件**: `src/components/Presets/PresetApplyDialog.tsx`

```typescript
import React, { useState, useEffect } from 'react'
import { Preset } from '@/core/types/Preset'
import { PresetApplyResult } from '@/core/types/PresetLoader'
import { usePresetLoader } from '@/hooks/usePresetLoader'

interface PresetApplyDialogProps {
  preset: Preset
  currentModelId: string
  onApply: () => void
  onCancel: () => void
}

export function PresetApplyDialog({
  preset,
  currentModelId,
  onApply,
  onCancel
}: PresetApplyDialogProps) {
  const { previewPreset, applyPreset } = usePresetLoader(currentModelId)
  const [preview, setPreview] = useState<PresetApplyResult | null>(null)
  const [applyMode, setApplyMode] = useState<'full' | 'partial' | 'merge'>('full')
  const [selectedParams, setSelectedParams] = useState<Set<string>>(new Set())

  useEffect(() => {
    // 预览应用结果
    const result = previewPreset(preset, { mode: applyMode })
    setPreview(result)

    // 默认选中所有可应用的参数
    if (applyMode === 'partial') {
      setSelectedParams(new Set(result.applied.map(p => p.id)))
    }
  }, [preset, applyMode, previewPreset])

  const handleApply = () => {
    if (applyMode === 'partial') {
      applyPreset(preset, {
        mode: 'partial',
        paramIds: Array.from(selectedParams)
      })
    } else {
      applyPreset(preset, { mode: applyMode })
    }
    onApply()
  }

  if (!preview) {
    return <div>加载中...</div>
  }

  return (
    <div className="preset-apply-dialog">
      <h3>应用预设：{preset.name}</h3>

      {/* 应用模式选择 */}
      <div className="apply-mode">
        <label>
          <input
            type="radio"
            checked={applyMode === 'full'}
            onChange={() => setApplyMode('full')}
          />
          完全应用（覆盖所有参数）
        </label>
        <label>
          <input
            type="radio"
            checked={applyMode === 'merge'}
            onChange={() => setApplyMode('merge')}
          />
          合并应用（保留已修改的参数）
        </label>
        <label>
          <input
            type="radio"
            checked={applyMode === 'partial'}
            onChange={() => setApplyMode('partial')}
          />
          部分应用（选择要应用的参数）
        </label>
      </div>

      {/* 预览结果 */}
      <div className="preview-results">
        {preview.applied.length > 0 && (
          <div className="preview-section">
            <h4>✓ 将应用 ({preview.applied.length})</h4>
            {preview.applied.map(param => (
              <div key={param.id} className="preview-item">
                {applyMode === 'partial' && (
                  <input
                    type="checkbox"
                    checked={selectedParams.has(param.id)}
                    onChange={e => {
                      const newSet = new Set(selectedParams)
                      if (e.target.checked) {
                        newSet.add(param.id)
                      } else {
                        newSet.delete(param.id)
                      }
                      setSelectedParams(newSet)
                    }}
                  />
                )}
                <span className="param-name">{param.id}</span>
                <span className="param-change">
                  {JSON.stringify(param.oldValue)} → {JSON.stringify(param.newValue)}
                </span>
              </div>
            ))}
          </div>
        )}

        {preview.conflicts.length > 0 && applyMode !== 'merge' && (
          <div className="preview-section warning">
            <h4>⚠ 冲突参数 ({preview.conflicts.length})</h4>
            <p>以下参数已被修改，应用预设将覆盖它们：</p>
            {preview.conflicts.map(param => (
              <div key={param.id} className="preview-item">
                <span className="param-name">{param.id}</span>
                <span className="param-change">
                  当前: {JSON.stringify(param.currentValue)} →
                  预设: {JSON.stringify(param.presetValue)}
                </span>
              </div>
            ))}
          </div>
        )}

        {preview.ignored.length > 0 && (
          <div className="preview-section info">
            <h4>ℹ 将忽略 ({preview.ignored.length})</h4>
            {preview.ignored.map(param => (
              <div key={param.id} className="preview-item">
                <span className="param-name">{param.id}</span>
                <span className="reason">{getIgnoreReasonText(param.reason)}</span>
              </div>
            ))}
          </div>
        )}

        {preview.invalid.length > 0 && (
          <div className="preview-section error">
            <h4>✗ 无效参数 ({preview.invalid.length})</h4>
            {preview.invalid.map(param => (
              <div key={param.id} className="preview-item">
                <span className="param-name">{param.id}</span>
                <span className="reason">{param.reason}</span>
              </div>
            ))}
          </div>
        )}
      </div>

      {/* 操作按钮 */}
      <div className="dialog-actions">
        <button onClick={onCancel}>取消</button>
        <button
          onClick={handleApply}
          disabled={
            applyMode === 'partial' && selectedParams.size === 0
          }
        >
          应用预设
        </button>
      </div>
    </div>
  )
}

function getIgnoreReasonText(reason: string): string {
  switch (reason) {
    case 'not-in-schema':
      return '参数不存在于当前模型'
    case 'global-preset-filter':
      return '全局预设不包含此参数'
    case 'not-selected':
      return '未选中'
    default:
      return reason
  }
}
```

## 风险与注意事项

### 风险

1. **性能问题**
   - 大量参数验证可能影响性能
   - 缓解：使用 Web Worker 进行验证

2. **复杂度增加**
   - 用户可能不理解各种应用模式
   - 缓解：提供清晰的说明和默认推荐

3. **撤销历史内存占用**
   - 保存多次撤销历史可能占用内存
   - 缓解：限制历史数量（最多 10 次）

### 注意事项

1. **用户体验**
   - 冲突提示应该友好，不要让用户困惑
   - 默认使用"完全应用"模式，简单直接
   - 高级用户可以选择其他模式

2. **数据安全**
   - 撤销功能只保存在内存，刷新页面会丢失
   - 重要操作前提示用户

3. **性能优化**
   - 预览结果应该缓存，避免重复计算
   - 参数验证结果应该缓存

## 回滚方案

如果新的加载逻辑出现问题：

1. **回退到简单加载**
   ```typescript
   // 简化版本：直接覆盖，不做验证
   const { setParams } = useModelParams(modelId)
   setParams(preset.params)
   ```

2. **禁用高级功能**
   - 禁用冲突检测
   - 禁用部分应用
   - 只保留完全应用模式

3. **移除验证器**
   - 如果验证器有 bug，可以临时禁用
   - 回退到只检查参数是否存在于 schema
