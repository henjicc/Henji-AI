# 任务 3-1-3：迁移历史数据

## 任务信息

- **阶段**: Phase 3 - 数据与存储
- **优先级**: P1（高优先级）
- **预计工时**: 6-8 小时
- **依赖**: 任务 3-1-2（数据库操作层）
- **前置条件**: DatabaseService 已实现并测试通过

## 目标

将现有的 `history.json` 数据安全地迁移到 SQLite 数据库，并清理 Base64 数据以减小存储大小。

## 背景

当前历史记录存储在 JSON 文件中，存在以下问题：

1. **Base64 膨胀**: 某些记录可能包含 `data:image/png;base64,...` 导致文件巨大
2. **查询效率低**: 需要加载整个文件才能查询
3. **数据完整性**: 缺乏验证和约束

**迁移关键原则**:
- ✅ 只保留 `filePath`，绝不保留 `url` 中的 base64 数据
- ✅ 验证文件路径有效性
- ✅ 保留完整备份
- ✅ 支持回滚

```typescript
import { readTextFile } from '@tauri-apps/plugin-fs'
import { appLocalDataDir, join } from '@tauri-apps/api/path'

interface LegacyHistoryItem {
  id: string
  provider: string
  model: string
  type: 'image' | 'video' | 'audio'
  prompt: string
  result: {
    url?: string
    filePath?: string
    taskId?: string
  }
  timestamp: number
  settings: Record<string, any>
}

export async function analyzeHistoryFile() {
  const appDataDir = await appLocalDataDir()
  const historyPath = await join(appDataDir, 'Henji-AI', 'history.json')

  try {
    const content = await readTextFile(historyPath)
    const data: LegacyHistoryItem[] = JSON.parse(content)

    console.log('========== 历史数据分析 ==========')
    console.log(`总记录数: ${data.length}`)
    console.log(`文件大小: ${(content.length / 1024 / 1024).toFixed(2)} MB`)

    // 统计 Base64 数据
    let base64Count = 0
    let base64Size = 0
    let validFilePathCount = 0
    let missingFileCount = 0

    for (const item of data) {
      if (item.result?.url?.startsWith('data:')) {
        base64Count++
        base64Size += item.result.url.length
      }

      if (item.result?.filePath) {
        validFilePathCount++
        // TODO: 验证文件是否存在
      } else {
        missingFileCount++
      }
    }

    console.log(`\nBase64 数据统计:`)
    console.log(`  包含 Base64 的记录: ${base64Count}`)
    console.log(`  Base64 数据大小: ${(base64Size / 1024 / 1024).toFixed(2)} MB`)
    console.log(`  占文件比例: ${((base64Size / content.length) * 100).toFixed(1)}%`)

    console.log(`\n文件路径统计:`)
    console.log(`  有效路径记录: ${validFilePathCount}`)
    console.log(`  缺失路径记录: ${missingFileCount}`)

    // 按类型统计
    const typeStats = data.reduce((acc, item) => {
      acc[item.type] = (acc[item.type] || 0) + 1
      return acc
    }, {} as Record<string, number>)

    console.log(`\n按类型统计:`)
    Object.entries(typeStats).forEach(([type, count]) => {
      console.log(`  ${type}: ${count}`)
    })

    // 按供应商统计
    const providerStats = data.reduce((acc, item) => {
      acc[item.provider] = (acc[item.provider] || 0) + 1
      return acc
    }, {} as Record<string, number>)

    console.log(`\n按供应商统计:`)
    Object.entries(providerStats).forEach(([provider, count]) => {
      console.log(`  ${provider}: ${count}`)
    })

    return {
      totalRecords: data.length,
      fileSize: content.length,
      base64Count,
      base64Size,
      validFilePathCount,
      missingFileCount,
      typeStats,
      providerStats,
    }
  } catch (error) {
    console.error('分析历史文件失败:', error)
    throw error
  }
}
```

### 步骤 2: 实现数据清洗逻辑

**文件**: `src/scripts/migration/cleanHistory.ts`

```typescript
import type { LegacyHistoryItem } from './types'
import type { HistoryRecord } from '@/services/database/types'

export function cleanHistoryItem(
  legacy: LegacyHistoryItem
): Omit<HistoryRecord, 'createdAt' | 'updatedAt'> | null {
  try {
    // 验证必需字段
    if (!legacy.id || !legacy.provider || !legacy.model || !legacy.type) {
      console.warn(`[Clean] 跳过无效记录: ${legacy.id}`)
      return null
    }

    // 提取文件路径（绝对不保留 Base64）
    let filePath: string | null = null

    if (legacy.result?.filePath) {
      // 清理文件路径
      filePath = legacy.result.filePath
        .replace(/\\/g, '/')  // 统一使用正斜杠
        .trim()

      // 验证路径格式
      if (!filePath || filePath.startsWith('data:')) {
        console.warn(`[Clean] 无效文件路径: ${legacy.id}`)
        filePath = null
      }
    }

    // 如果没有有效文件路径，跳过此记录
    if (!filePath) {
      console.warn(`[Clean] 跳过无文件路径的记录: ${legacy.id}`)
      return null
    }

    // 清理参数（移除敏感数据）
    const params = { ...legacy.settings }

    // 移除可能包含 Base64 的字段
    delete params.uploadedImages
    delete params.uploadedVideos
    delete params.imageDataUrl

    // 转换为新格式
    return {
      id: legacy.id,
      providerId: legacy.provider,
      modelId: legacy.model,
      type: legacy.type,
      prompt: legacy.prompt || null,
      params,
      filePath,
      taskId: legacy.result?.taskId || null,
      status: 'completed',
      errorMessage: null,
      cost: null,  // 旧数据没有成本信息
      duration: null,  // 旧数据没有时长信息
    }
  } catch (error) {
    console.error(`[Clean] 清洗记录失败: ${legacy.id}`, error)
    return null
  }
}
```

### 步骤 3: 实现迁移脚本

**文件**: `src/scripts/migration/migrateHistory.ts`

```typescript
import { readTextFile, writeTextFile, exists, copyFile } from '@tauri-apps/plugin-fs'
import { appLocalDataDir, join } from '@tauri-apps/api/path'
import { databaseService } from '@/services/database/DatabaseService'
import { cleanHistoryItem } from './cleanHistory'
import type { LegacyHistoryItem } from './types'

export interface MigrationResult {
  success: boolean
  totalRecords: number
  migratedRecords: number
  skippedRecords: number
  errors: string[]
  backupPath: string
  originalSize: number
  newSize: number
}

export async function migrateHistoryToDatabase(): Promise<MigrationResult> {
  const result: MigrationResult = {
    success: false,
    totalRecords: 0,
    migratedRecords: 0,
    skippedRecords: 0,
    errors: [],
    backupPath: '',
    originalSize: 0,
    newSize: 0,
  }

  const appDataDir = await appLocalDataDir()
  const historyPath = await join(appDataDir, 'Henji-AI', 'history.json')
  const backupPath = await join(
    appDataDir,
    'Henji-AI',
    `history.backup.${Date.now()}.json`
  )

  try {
    // 步骤 1: 检查文件是否存在
    const fileExists = await exists(historyPath)
    if (!fileExists) {
      console.log('[Migration] 历史文件不存在，跳过迁移')
      result.success = true
      return result
    }

    // 步骤 2: 备份原文件
    console.log('[Migration] 备份历史文件...')
    await copyFile(historyPath, backupPath)
    result.backupPath = backupPath
    console.log(`[Migration] 备份完成: ${backupPath}`)

    // 步骤 3: 读取并解析
    console.log('[Migration] 读取历史文件...')
    const content = await readTextFile(historyPath)
    result.originalSize = content.length

    const legacyData: LegacyHistoryItem[] = JSON.parse(content)
    result.totalRecords = legacyData.length
    console.log(`[Migration] 读取到 ${result.totalRecords} 条记录`)

    // 步骤 4: 初始化数据库
    console.log('[Migration] 初始化数据库...')
    await databaseService.init()

    // 步骤 5: 清洗并迁移数据
    console.log('[Migration] 开始迁移数据...')
    for (let i = 0; i < legacyData.length; i++) {
      const legacy = legacyData[i]

      if (i % 100 === 0) {
        console.log(`[Migration] 进度: ${i}/${result.totalRecords}`)
      }

      try {
        const cleaned = cleanHistoryItem(legacy)

        if (!cleaned) {
          result.skippedRecords++
          continue
        }

        await databaseService.insertHistory(cleaned)
        result.migratedRecords++
      } catch (error: any) {
        result.errors.push(`记录 ${legacy.id}: ${error.message}`)
        result.skippedRecords++
      }
    }

    // 步骤 6: 验证迁移结果
    console.log('[Migration] 验证迁移结果...')
    const migratedCount = (
      await databaseService.getHistory({ limit: 1000000 })
    ).length

    if (migratedCount !== result.migratedRecords) {
      throw new Error(
        `迁移数量不匹配: 预期 ${result.migratedRecords}, 实际 ${migratedCount}`
      )
    }

    // 步骤 7: 删除旧文件（保留备份）
    console.log('[Migration] 删除旧历史文件...')
    // 写入空数组，保留文件结构
    await writeTextFile(historyPath, JSON.stringify([]))
    result.newSize = 2  // "[]"

    result.success = true
    console.log('[Migration] 迁移完成！')
    console.log(`  总记录: ${result.totalRecords}`)
    console.log(`  成功迁移: ${result.migratedRecords}`)
    console.log(`  跳过记录: ${result.skippedRecords}`)
    console.log(`  错误数: ${result.errors.length}`)
    console.log(`  文件大小: ${(result.originalSize / 1024 / 1024).toFixed(2)} MB -> ${(result.newSize / 1024).toFixed(2)} KB`)
    console.log(`  压缩比: ${((1 - result.newSize / result.originalSize) * 100).toFixed(1)}%`)

    return result
  } catch (error: any) {
    console.error('[Migration] 迁移失败:', error)
    result.errors.push(error.message)

    // 尝试恢复备份
    if (result.backupPath) {
      console.log('[Migration] 尝试从备份恢复...')
      try {
        await copyFile(backupPath, historyPath)
        console.log('[Migration] 恢复成功')
      } catch (restoreError) {
        console.error('[Migration] 恢复失败:', restoreError)
      }
    }

    throw error
  }
}
```

### 步骤 4: 创建迁移 UI 组件

**文件**: `src/components/Migration/HistoryMigrationDialog.tsx`

```tsx
import { useState } from 'react'
import { migrateHistoryToDatabase } from '@/scripts/migration/migrateHistory'
import type { MigrationResult } from '@/scripts/migration/migrateHistory'

export function HistoryMigrationDialog() {
  const [isOpen, setIsOpen] = useState(true)
  const [isMigrating, setIsMigrating] = useState(false)
  const [result, setResult] = useState<MigrationResult | null>(null)

  const handleMigrate = async () => {
    setIsMigrating(true)
    try {
      const migrationResult = await migrateHistoryToDatabase()
      setResult(migrationResult)
    } catch (error) {
      console.error('迁移失败:', error)
    } finally {
      setIsMigrating(false)
    }
  }

  if (!isOpen) return null

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
      <div className="bg-white dark:bg-gray-800 rounded-lg p-6 max-w-md w-full">
        <h2 className="text-xl font-bold mb-4">数据库迁移</h2>

        {!result && (
          <>
            <p className="text-gray-600 dark:text-gray-300 mb-4">
              检测到旧版本的历史记录文件，需要迁移到新的数据库格式。
            </p>
            <p className="text-sm text-gray-500 mb-6">
              此操作将：
              <br />• 备份现有数据
              <br />• 清理 Base64 数据以减小存储
              <br />• 迁移到 SQLite 数据库
              <br />• 支持更快的查询和筛选
            </p>

            <div className="flex gap-3">
              <button
                onClick={handleMigrate}
                disabled={isMigrating}
                className="flex-1 bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600 disabled:opacity-50"
              >
                {isMigrating ? '迁移中...' : '开始迁移'}
              </button>
              <button
                onClick={() => setIsOpen(false)}
                disabled={isMigrating}
                className="px-4 py-2 border rounded hover:bg-gray-50 dark:hover:bg-gray-700"
              >
                稍后
              </button>
            </div>
          </>
        )}

        {result && (
          <>
            {result.success ? (
              <div className="text-green-600 mb-4">
                ✓ 迁移成功！
              </div>
            ) : (
              <div className="text-red-600 mb-4">
                ✗ 迁移失败
              </div>
            )}

            <div className="text-sm space-y-2 mb-6">
              <div>总记录数: {result.totalRecords}</div>
              <div>成功迁移: {result.migratedRecords}</div>
              <div>跳过记录: {result.skippedRecords}</div>
              <div>
                文件大小: {(result.originalSize / 1024 / 1024).toFixed(2)} MB
                → {(result.newSize / 1024).toFixed(2)} KB
              </div>
              <div>
                节省空间: {((1 - result.newSize / result.originalSize) * 100).toFixed(1)}%
              </div>

              {result.errors.length > 0 && (
                <details className="mt-4">
                  <summary className="cursor-pointer text-red-600">
                    查看错误 ({result.errors.length})
                  </summary>
                  <div className="mt-2 max-h-40 overflow-auto text-xs">
                    {result.errors.map((err, i) => (
                      <div key={i} className="text-red-500">
                        {err}
                      </div>
                    ))}
                  </div>
                </details>
              )}
            </div>

            <button
              onClick={() => setIsOpen(false)}
              className="w-full bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
            >
              完成
            </button>
          </>
        )}
      </div>
    </div>
  )
}
```

### 步骤 5: 集成到应用启动流程

**文件**: `src/App.tsx`

```typescript
import { useEffect, useState } from 'react'
import { exists } from '@tauri-apps/plugin-fs'
import { appLocalDataDir, join } from '@tauri-apps/api/path'
import { HistoryMigrationDialog } from '@/components/Migration/HistoryMigrationDialog'

export function App() {
  const [needsMigration, setNeedsMigration] = useState(false)

  useEffect(() => {
    checkMigrationNeeded()
  }, [])

  async function checkMigrationNeeded() {
    const appDataDir = await appLocalDataDir()
    const historyPath = await join(appDataDir, 'Henji-AI', 'history.json')
    const dbPath = await join(appDataDir, 'Henji-AI', 'database.db')

    const hasHistoryFile = await exists(historyPath)
    const hasDatabase = await exists(dbPath)

    // 如果有历史文件但没有数据库，需要迁移
    setNeedsMigration(hasHistoryFile && !hasDatabase)
  }

  return (
    <>
      {needsMigration && <HistoryMigrationDialog />}
      {/* 应用其他内容 */}
    </>
  )
}
```
