## 目标

实现图片和视频上传组件，支持预览、编辑、格式转换和智能匹配

## 背景

- ImageUpload（图片上传）
- VideoUpload（视频上传）

这些组件需要：
- 支持拖拽上传
- 支持多文件上传（有数量限制）
- 支持预览和删除
- 支持格式转换（Base64/URL）
- 触发智能匹配（如自动设置分辨率）
- 集成图片编辑器

```typescript
// 智能匹配工具
export function calculateAspectRatio(width: number, height: number): string {
  const ratio = width / height
  const ratios = {
    '16:9': 16/9,
    '9:16': 9/16,
    '1:1': 1,
    '4:3': 4/3,
    '3:4': 3/4
  }

  let closest = '16:9'
  let minDiff = Math.abs(ratio - ratios['16:9'])

  for (const [key, value] of Object.entries(ratios)) {
    const diff = Math.abs(ratio - value)
    if (diff < minDiff) {
      minDiff = diff
      closest = key
    }
  }

  return closest
}
```

## 风险与注意事项

### 风险
- 大文件上传可能导致内存溢出
- Base64 编码会增大文件体积
- 跨域问题导致 URL 转 Base64 失败

### 注意事项
- 使用 FileReader API 读取文件
- 限制 Base64 图片最大尺寸（建议 5MB）
- 视频上传使用 CDN，不使用 Base64
- 处理 EXIF 旋转问题
- 移动端考虑压缩图片
- 上传失败提供重试机制
- 清理未使用的上传文件（内存管理）
- 图片编辑后保持原格式和质量

### ImageUpload 示例

```typescript
import React, { useState } from 'react'
import { I18nText } from '@/core/types'
import { useI18n } from '@/hooks/useI18n'
import { UploadArea } from './UploadArea'
import { FilePreview } from './FilePreview'
import { ImageEditor } from '@/components/ImageEditor'
import { fileToBase64 } from '@/utils/fileConverter'
import { calculateAspectRatio } from '@/utils/smartMatch'

interface ImageUploadProps {
  value: string[]
  onChange: (value: string[]) => void
  label: I18nText
  maxCount: number
  format: 'base64' | 'url'
  base64Prefix?: boolean
  onSmartMatch?: (metadata: { aspectRatio: string; width: number; height: number }) => void
  disabled?: boolean
}

export const ImageUpload: React.FC<ImageUploadProps> = ({
  value,
  onChange,
  label,
  maxCount,
  format,
  base64Prefix = false,
  onSmartMatch,
  disabled = false
}) => {
  const { t } = useI18n()
  const [editingIndex, setEditingIndex] = useState<number | null>(null)

  const handleUpload = async (file: File) => {
    if (value.length >= maxCount) {
      alert(`最多上传 ${maxCount} 张图片`)
      return
    }

    // 读取图片元数据
    const img = await loadImage(file)
    const aspectRatio = calculateAspectRatio(img.width, img.height)

    // 转换格式
    let imageData: string
    if (format === 'base64') {
      const base64 = await fileToBase64(file)
      imageData = base64Prefix ? base64 : base64.split(',')[1]
    } else {
      // 上传到服务器获取 URL
      imageData = await uploadToServer(file)
    }

    onChange([...value, imageData])

    // 触发智能匹配
    if (onSmartMatch) {
      onSmartMatch({
        aspectRatio,
        width: img.width,
        height: img.height
      })
    }
  }

  const handleDelete = (index: number) => {
    onChange(value.filter((_, i) => i !== index))
  }

  const handleEdit = (index: number) => {
    setEditingIndex(index)
  }

  const handleEditComplete = (editedImageData: string) => {
    const newValue = [...value]
    newValue[editingIndex!] = editedImageData
    onChange(newValue)
    setEditingIndex(null)
  }

  return (
    <div className="param-image-upload">
      <label>{t(label)}</label>

      <div className="upload-container">
        {value.map((url, index) => (
          <FilePreview
            key={index}
            type="image"
            url={format === 'base64' && !base64Prefix ? `data:image/png;base64,${url}` : url}
            onDelete={() => handleDelete(index)}
            onEdit={() => handleEdit(index)}
          />
        ))}

        {value.length < maxCount && (
          <UploadArea
            accept={['image/png', 'image/jpeg', 'image/webp']}
            maxSize={10}  // 10MB
            onUpload={handleUpload}
            disabled={disabled}
          />
        )}
      </div>

      {editingIndex !== null && (
        <ImageEditor
          imageData={value[editingIndex]}
          onSave={handleEditComplete}
          onCancel={() => setEditingIndex(null)}
        />
      )}
    </div>
  )
}

function loadImage(file: File): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    img.onload = () => resolve(img)
    img.onerror = reject
    img.src = URL.createObjectURL(file)
  })
}
```

### UploadArea 示例

```typescript
import React, { useRef, useState } from 'react'

interface UploadAreaProps {
  accept: string[]
  maxSize?: number
  onUpload: (file: File) => Promise<void>
  disabled?: boolean
}

export const UploadArea: React.FC<UploadAreaProps> = ({
  accept,
  maxSize = 10,
  onUpload,
  disabled = false
}) => {
  const inputRef = useRef<HTMLInputElement>(null)
  const [isDragging, setIsDragging] = useState(false)
  const [uploading, setUploading] = useState(false)

  const handleDrop = async (e: React.DragEvent) => {
    e.preventDefault()
    setIsDragging(false)

    if (disabled || uploading) return

    const file = e.dataTransfer.files[0]
    if (file) {
      await processFile(file)
    }
  }

  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (file) {
      await processFile(file)
    }
  }

  const processFile = async (file: File) => {
    // 验证文件类型
    if (!accept.includes(file.type)) {
      alert(`不支持的文件类型: ${file.type}`)
      return
    }

    // 验证文件大小
    if (maxSize && file.size > maxSize * 1024 * 1024) {
      alert(`文件大小超过 ${maxSize}MB`)
      return
    }

    setUploading(true)
    try {
      await onUpload(file)
    } catch (error) {
      console.error('Upload failed:', error)
      alert('上传失败，请重试')
    } finally {
      setUploading(false)
    }
  }

  return (
    <div
      className={`upload-area ${isDragging ? 'dragging' : ''} ${uploading ? 'uploading' : ''}`}
      onDragOver={(e) => {
        e.preventDefault()
        setIsDragging(true)
      }}
      onDragLeave={() => setIsDragging(false)}
      onDrop={handleDrop}
      onClick={() => !disabled && !uploading && inputRef.current?.click()}
    >
      <input
        ref={inputRef}
        type="file"
        accept={accept.join(',')}
        onChange={handleFileSelect}
        style={{ display: 'none' }}
      />

      {uploading ? (
        <div className="upload-progress">上传中...</div>
      ) : (
        <div className="upload-hint">
          <span>拖拽文件到此处或点击上传</span>
          <span className="upload-limit">支持 {accept.map(t => t.split('/')[1]).join(', ')} 格式，最大 {maxSize}MB</span>
        </div>
      )}
    </div>
  )
}
```
