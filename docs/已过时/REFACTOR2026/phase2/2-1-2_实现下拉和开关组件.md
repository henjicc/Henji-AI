# 2-1-2 实现下拉和开关组件

## 目标

实现下拉选择（Dropdown）、开关（Switch）、单选（Radio）等选择类组件

## 背景

选择类组件是参数配置中最常用的组件类型，需要实现：
- DropdownInput（下拉选择）
- SwitchInput（开关切换）
- RadioInput（单选组）

这些组件需要：
- 支持选项过滤（联动系统）
- 支持动态选项加载
- 支持搜索和键盘导航（Dropdown）
- 良好的视觉反馈

- 下拉面板定位复杂，可能在某些场景下溢出

- 移动端考虑使用原生 select（可选）
- 下拉面板自动翻转（上方空间不足时向上展开）

## 参考实现

### DropdownInput 示例

```typescript
import React, { useState, useRef, useEffect } from 'react'
import { createPortal } from 'react-dom'
import { I18nText } from '@/core/types'
import { useI18n } from '@/hooks/useI18n'
import { useDropdown } from '@/hooks/useDropdown'
import { useKeyboardNav } from '@/hooks/useKeyboardNav'

interface DropdownInputProps {
  value: string | number
  onChange: (value: string | number) => void
  label: I18nText
  options: Array<{ value: string | number; label: I18nText; disabled?: boolean }>
  searchable?: boolean
  disabled?: boolean
}

export const DropdownInput: React.FC<DropdownInputProps> = ({
  value,
  onChange,
  label,
  options,
  searchable = false,
  disabled = false
}) => {
  const { t } = useI18n()
  const triggerRef = useRef<HTMLDivElement>(null)
  const { isOpen, toggle, close, panelPosition } = useDropdown(triggerRef)
  const [searchText, setSearchText] = useState('')

  const filteredOptions = searchable
    ? options.filter(opt => t(opt.label).toLowerCase().includes(searchText.toLowerCase()))
    : options

  const { focusedIndex, handleKeyDown } = useKeyboardNav({
    items: filteredOptions,
    onSelect: (opt) => {
      onChange(opt.value)
      close()
    },
    isOpen
  })

  const selectedOption = options.find(opt => opt.value === value)

  return (
    <div className="param-dropdown">
      <label>{t(label)}</label>
      <div
        ref={triggerRef}
        className={`dropdown-trigger ${isOpen ? 'open' : ''}`}
        onClick={() => !disabled && toggle()}
        onKeyDown={handleKeyDown}
        tabIndex={0}
      >
        <span>{selectedOption ? t(selectedOption.label) : 'Select...'}</span>
        <span className="dropdown-arrow">▼</span>
      </div>

      {isOpen && createPortal(
        <div
          className="dropdown-panel"
          style={{
            position: 'absolute',
            top: panelPosition.top,
            left: panelPosition.left,
            width: panelPosition.width
          }}
        >
          {searchable && (
            <input
              type="text"
              className="dropdown-search"
              placeholder="Search..."
              value={searchText}
              onChange={(e) => setSearchText(e.target.value)}
              autoFocus
            />
          )}
          <div className="dropdown-options">
            {filteredOptions.map((opt, index) => (
              <div
                key={opt.value}
                className={`dropdown-option ${opt.value === value ? 'selected' : ''} ${index === focusedIndex ? 'focused' : ''} ${opt.disabled ? 'disabled' : ''}`}
                onClick={() => {
                  if (!opt.disabled) {
                    onChange(opt.value)
                    close()
                  }
                }}
              >
                {t(opt.label)}
              </div>
            ))}
          </div>
        </div>,
        document.body
      )}
    </div>
  )
}
```

### SwitchInput 示例

```typescript
import React from 'react'
import { I18nText } from '@/core/types'
import { useI18n } from '@/hooks/useI18n'

interface SwitchInputProps {
  value: boolean
  onChange: (value: boolean) => void
  label: I18nText
  description?: I18nText
  disabled?: boolean
}

export const SwitchInput: React.FC<SwitchInputProps> = ({
  value,
  onChange,
  label,
  description,
  disabled = false
}) => {
  const { t } = useI18n()

  return (
    <div className="param-switch">
      <div className="switch-label-group">
        <label>{t(label)}</label>
        {description && <span className="switch-description">{t(description)}</span>}
      </div>
      <button
        className={`switch-button ${value ? 'on' : 'off'}`}
        onClick={() => !disabled && onChange(!value)}
        disabled={disabled}
        role="switch"
        aria-checked={value}
      >
        <span className="switch-slider" />
      </button>
    </div>
  )
}
```
