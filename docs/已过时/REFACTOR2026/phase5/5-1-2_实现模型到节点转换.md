```typescript
paramsToInputPorts(params: ParamDef[]): InputPort[] {
  const ports: InputPort[] = []

  for (const param of params) {
    // 跳过纯 UI 参数（如分隔符）
    if (param.component === 'divider') continue

    // 基础端口
    const port: InputPort = {
      id: param.id,
      name: param.name,
      type: this.paramTypeToPortType(param),
      required: param.required ?? false,
      default: param.default,
      description: param.tooltip
    }

    ports.push(port)

    // 处理复合参数（如 resolution 分解为 width, height）
    if (this.isCompositeParam(param)) {
      ports.push(...this.expandCompositeParam(param))
    }
  }

  return ports
}

private paramTypeToPortType(param: ParamDef): PortDataType {
  switch (param.component) {
    case 'text':
    case 'dropdown':
      return 'string'
    case 'number':
    case 'slider':
      return 'number'
    case 'switch':
      return 'boolean'
    case 'image-upload':
      return 'image'
    case 'video-upload':
      return 'video'
    case 'panel':
      // 根据面板类型决定
      if (param.panelType === 'resolution') return 'object'
      return 'any'
    default:
      return 'any'
  }
}
```

```typescript
getOutputPorts(modelType: 'image' | 'video' | 'audio'): OutputPort[] {
  const baseOutput: OutputPort = {
    id: 'output',
    name: this.getOutputName(modelType),
    type: modelType,
    description: {
      zh: `生成的${this.getTypeLabel(modelType)}`,
      en: `Generated ${modelType}`
    }
  }

  const metadataOutput: OutputPort = {
    id: 'metadata',
    name: { zh: '元数据', en: 'Metadata' },
    type: 'object',
    description: { zh: '生成任务的元数据', en: 'Task metadata' }
  }

  return [baseOutput, metadataOutput]
}

private getOutputName(type: string): I18nText {
  const names = {
    image: { zh: '图片', en: 'Image' },
    video: { zh: '视频', en: 'Video' },
    audio: { zh: '音频', en: 'Audio' }
  }
  return names[type] || { zh: '输出', en: 'Output' }
}
```

```typescript
private createExecutor(model: ModelDefinition): NodeExecutor {
  return async (inputs: Record<string, any>, context?: ExecutionContext) => {
    try {
      // 1. 获取适配器
      const adapter = getAdapter(model.meta.provider)

      // 2. 构建请求参数
      const params = this.inputsToParams(inputs, model.params)

      // 3. 调用生成方法
      let result: any
      const generateMethod = this.getGenerateMethod(model.meta.type)

      if (model.meta.type === 'image') {
        result = await adapter.generateImage({
          ...params,
          model: model.meta.id,
          onProgress: context?.onProgress
        })
      } else if (model.meta.type === 'video') {
        result = await adapter.generateVideo({
          ...params,
          model: model.meta.id,
          onProgress: context?.onProgress
        })
      } else if (model.meta.type === 'audio') {
        result = await adapter.generateAudio({
          ...params,
          model: model.meta.id,
          onProgress: context?.onProgress
        })
      }

      // 4. 返回节点输出
      return {
        output: result.url || result.filePath,
        metadata: {
          taskId: result.taskId,
          status: result.status,
          filePath: result.filePath
        }
      }
    } catch (error) {
      console.error(`Node execution failed:`, error)
      return {
        output: null,
        metadata: {
          status: 'failed',
          error: error.message
        }
      }
    }
  }
}

private inputsToParams(
  inputs: Record<string, any>,
  paramDefs: ParamDef[]
): Record<string, any> {
  const params: Record<string, any> = {}

  for (const paramDef of paramDefs) {
    const value = inputs[paramDef.id]
    if (value !== undefined) {
      params[paramDef.id] = value
    } else if (paramDef.default !== undefined) {
      params[paramDef.id] = paramDef.default
    }
  }

  return params
}
```

```typescript
modelToNode(model: ModelDefinition): ModelNode {
  // 检查缓存
  if (this.cache.has(model.meta.id)) {
    return this.cache.get(model.meta.id)!
  }

  // 转换
  const node: ModelNode = {
    id: `node-${model.meta.id}`,
    type: 'model',
    modelId: model.meta.id,

    meta: {
      name: model.meta.name,
      description: model.meta.description,
      icon: model.meta.icon,
      category: model.meta.type,
      tags: model.meta.tags
    },

    inputs: this.paramsToInputPorts(model.params),
    outputs: this.getOutputPorts(model.meta.type),

    execute: this.createExecutor(model)
  }

  // 缓存
  this.cache.set(model.meta.id, node)

  return node
}
```

```typescript
convertAllModels(filter?: {
  provider?: string
  type?: 'image' | 'video' | 'audio'
  tags?: string[]
}): ModelNode[] {
  const registry = ModelRegistry.getInstance()
  let models = registry.listAllModels()

  // 应用过滤
  if (filter?.provider) {
    models = models.filter(m => m.meta.provider === filter.provider)
  }
  if (filter?.type) {
    models = models.filter(m => m.meta.type === filter.type)
  }
  if (filter?.tags) {
    models = models.filter(m =>
      filter.tags!.some(tag => m.meta.tags?.includes(tag))
    )
  }

  return models.map(model => this.modelToNode(model))
}
```

```typescript
private expandCompositeParam(param: ParamDef): InputPort[] {
  // 例如：resolution 参数展开为 width, height
  if (param.panelType === 'resolution') {
    return [
      {
        id: `${param.id}.width`,
        name: { zh: '宽度', en: 'Width' },
        type: 'number',
        required: false
      },
      {
        id: `${param.id}.height`,
        name: { zh: '高度', en: 'Height' },
        type: 'number',
        required: false
      }
    ]
  }
  return []
}

validateNode(node: ModelNode): { valid: boolean; errors: string[] } {
  const errors: string[] = []

  // 检查必需端口
  for (const input of node.inputs) {
    if (input.required && input.default === undefined) {
      errors.push(`Required input port "${input.id}" has no default value`)
    }
  }

  // 检查执行器
  if (typeof node.execute !== 'function') {
    errors.push('Node execute is not a function')
  }

  return { valid: errors.length === 0, errors }
}
```

## 风险与注意事项

### 风险
- 复合参数的展开逻辑可能不完善
- 不同模型的参数差异可能导致转换失败
- 节点执行时的错误处理可能不够健全

### 注意事项
- 缓存机制应考虑模型配置变化的情况
- 参数映射时要处理类型转换
- 执行器应捕获所有可能的异常
- 考虑节点执行的取消功能（使用 AbortSignal）
- 日志输出应使用统一的日志工具
- 复合参数的处理策略需要明确（展开 vs 保持对象）

## 实现策略建议

### 复合参数处理

两种策略：

**策略 A：完全展开**
```typescript
// resolution 展开为独立端口
inputs: [
  { id: 'resolution.width', type: 'number' },
  { id: 'resolution.height', type: 'number' },
  { id: 'resolution.aspectRatio', type: 'string' },
  { id: 'resolution.quality', type: 'string' }
]
```

**策略 B：保持对象**
```typescript
// resolution 保持为单个对象端口
inputs: [
  {
    id: 'resolution',
    type: 'object',
    schema: { width: 'number', height: 'number', ... }
  }
]
```

**建议**：优先使用策略 B（保持对象），简化连接逻辑。在画布 UI 中可以展开显示。

### 执行器优化

```typescript
// 添加超时控制
private createExecutor(model: ModelDefinition): NodeExecutor {
  return async (inputs, context) => {
    const timeout = model.meta.polling?.maxAttempts
      ? model.meta.polling.maxAttempts * model.meta.polling.interval
      : 300000 // 默认 5 分钟

    const timeoutPromise = new Promise((_, reject) => {
      setTimeout(() => reject(new Error('Node execution timeout')), timeout)
    })

    const executePromise = this.executeNode(model, inputs, context)

    return Promise.race([executePromise, timeoutPromise])
  }
}
```
